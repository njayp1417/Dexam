Questions

  1. Develop a standalone factorial program in Assembly language.
  2. Convert your solution in (1) above to a function that is called from a main program.

1. Standalone Factorial Program

.MODEL SMALL
.STACK 100H

.DATA
NUM     DB 5
RESULT  DW ?

.CODE
MAIN PROC

    MOV AX, @DATA
    MOV DS, AX

    MOV AL, NUM
    MOV AH, 0

    MOV CX, AX
    MOV AX, 1

FACTORIAL_LOOP:
    MUL CX
    LOOP FACTORIAL_LOOP

    MOV RESULT, AX

    MOV AH, 4CH
    INT 21H

MAIN ENDP
END MAIN

2. Factorial as a Function Called from Main

.MODEL SMALL
.STACK 100H

.DATA
NUM     DB 5
RESULT  DW ?

.CODE

MAIN PROC

    MOV AX, @DATA
    MOV DS, AX

    MOV AL, NUM
    MOV AH, 0

    CALL FACTORIAL

    MOV RESULT, AX

    MOV AH, 4CH
    INT 21H

MAIN ENDP


FACTORIAL PROC

    MOV CX, AX
    MOV AX, 1

FACT_LOOP:
    MUL CX
    LOOP FACT_LOOP

    RET

FACTORIAL ENDP

END MAIN
