1. Standalone Factorial Program

; Standalone Factorial Program
; Computes n! where n = 5
; Assembler: NASM (x86, 32-bit)

section .data
    n       dd  5               ; input: compute 5!
    result  dd  0               ; store final result

section .text
    global _start

_start:
    mov     ecx, [n]            ; ECX = n (loop counter)
    mov     eax, 1              ; EAX = 1 (accumulator)

    cmp     ecx, 0              ; if n == 0, 0! = 1
    je      done

factorial_loop:
    mul     ecx                 ; EAX = EAX * ECX
    dec     ecx                 ; ECX = ECX - 1
    jnz     factorial_loop      ; repeat while ECX != 0

done:
    mov     [result], eax       ; store result (120) in memory

    mov     eax, 1              ; syscall: exit
    xor     ebx, ebx
    int     0x80

2. Factorial as a Function Called from Main

; Factorial as a Procedure Called from Main
; Assembler: NASM (x86, 32-bit)

section .data
    n       dd  6               ; input: compute 6!
    result  dd  0               ; store final result

section .text
    global _start

; PROCEDURE: factorial
;   Input  : ECX = n
;   Output : EAX = n!
factorial:
    mov     eax, 1              ; EAX = 1 (accumulator)

    cmp     ecx, 0              ; if n == 0, return 1
    je      .done

.loop:
    mul     ecx                 ; EAX = EAX * ECX
    dec     ecx                 ; ECX--
    jnz     .loop               ; repeat while ECX != 0

.done:
    ret                         ; return to caller; result in EAX

; MAIN PROGRAM
_start:
    mov     ecx, [n]            ; load n into ECX
    call    factorial           ; call the factorial procedure
    mov     [result], eax       ; save result (720) in memory

    mov     eax, 1              ; syscall: exit
    xor     ebx, ebx
    int     0x80
